import OBR from "https://esm.sh/@owlbear-rodeo/sdk@3.1.0";
import { CombatStore, SYNC_KEY } from "./store.js";
import { renderizarCombate, popularSelectFichas } from "./ui.js";

OBR.onReady(async () => {
  const listContainer = document.getElementById("combat-list-container");
  const atualizarTela = () => renderizarCombate(listContainer, CombatStore);

  CombatStore.registrarCallback(atualizarTela);
  CombatStore.carregar();

  // 1. Puxa estado inicial da sala
  try {
    const meta = await OBR.room.getMetadata();
    if (meta?.[SYNC_KEY]) {
      CombatStore.mesclarRemoto(meta[SYNC_KEY]);
    }
  } catch (_) {}

  // 2. Ouve mudanças de Room Metadata e Broadcast em tempo real
  OBR.room.onMetadataChange((meta) => {
    if (meta?.[SYNC_KEY]) {
      CombatStore.mesclarRemoto(meta[SYNC_KEY]);
    }
  });

  OBR.broadcast.onMessage("solaria/combate-sync", (e) => {
    if (e.data) {
      CombatStore.mesclarRemoto(e.data);
    }
  });

  // 3. Controles Principais
  document.getElementById("btn-proximo-turno").addEventListener("click", () => {
    CombatStore.avancarTurno();
  });

  document.getElementById("btn-fechar-combate").addEventListener("click", async () => {
    try {
      await OBR.broadcast.sendMessage("solaria/combate-fechado", true, { destination: "LOCAL" });
    } catch (_) {}
    await OBR.popover.close("solaria-painel-combate");
  });

  document.getElementById("btn-limpar-lista").addEventListener("click", () => {
    if (window.confirm("Deseja zerar a lista de iniciativa e reiniciar no Turno 1?")) {
      CombatStore.limpar();
    }
  });

  // 4. Gaveta de Adição
  const drawer = document.getElementById("add-drawer");
  const selFicha = document.getElementById("sel-ficha-existente");
  const inpNome = document.getElementById("inp-custom-nome");
  const inpInit = document.getElementById("inp-custom-init");

  document.getElementById("btn-toggle-add").addEventListener("click", async () => {
    const abriu = drawer.classList.toggle("is-open");
    if (abriu) {
      await popularSelectFichas(selFicha, OBR);
    }
  });

  document.getElementById("btn-fechar-drawer").addEventListener("click", () => {
    drawer.classList.remove("is-open");
  });

  document.getElementById("btn-confirmar-add").addEventListener("click", () => {
    let nome = inpNome.value.trim();
    let init = parseInt(inpInit.value, 10);
    let fichaId = null;

    if (selFicha.value) {
      try {
        const f = JSON.parse(selFicha.value);
        if (!nome) nome = f.nome;
        if (isNaN(init)) init = f.iniciativa;
        fichaId = f.id;
      } catch (_) {}
    }

    if (!nome) {
      inpNome.focus();
      return;
    }

    if (isNaN(init)) {
      init = Math.floor(Math.random() * 20) + 1;
    }

    CombatStore.adicionarParticipante(nome, init, fichaId);
    drawer.classList.remove("is-open");
    inpNome.value = "";
    inpInit.value = "";
    selFicha.value = "";
  });

  atualizarTela();
});

import OBR from "https://esm.sh/@owlbear-rodeo/sdk@3.1.0";

export const SYNC_KEY = "solaria/sync_combate";
const STORAGE_KEY = "solaria_combate_estado";

export const CombatStore = {
  rodada: 1,
  ativoIndex: 0,
  participantes: [],
  onChangeCallback: null,

  registrarCallback(cb) {
    this.onChangeCallback = cb;
  },

  notificarUI() {
    if (typeof this.onChangeCallback === "function") {
      this.onChangeCallback();
    }
  },

  carregar() {
    try {
      const salvo = JSON.parse(localStorage.getItem(STORAGE_KEY) || "null");
      if (salvo && Array.isArray(salvo.participantes)) {
        this.rodada = salvo.rodada || 1;
        this.ativoIndex = salvo.ativoIndex || 0;
        this.participantes = salvo.participantes;
      }
    } catch (_) {}
    return this.obterPayload();
  },

  obterPayload() {
    return {
      rodada: this.rodada,
      ativoIndex: this.ativoIndex,
      participantes: this.participantes,
      atualizadoEm: Date.now()
    };
  },

  salvarLocal() {
    localStorage.setItem(STORAGE_KEY, JSON.stringify(this.obterPayload()));
  },

  async sincronizarRede() {
    this.salvarLocal();
    this.notificarUI();
    const payload = this.obterPayload();
    try {
      await OBR.room.setMetadata({ [SYNC_KEY]: payload });
      await OBR.broadcast.sendMessage("solaria/combate-sync", payload);
    } catch (err) {
      console.error("Erro ao sincronizar combate:", err);
    }
  },

  mesclarRemoto(dados) {
    if (!dados || !Array.isArray(dados.participantes)) return;
    this.rodada = dados.rodada || 1;
    this.ativoIndex = dados.ativoIndex || 0;
    this.participantes = dados.participantes;
    this.salvarLocal();
    this.notificarUI();
  },

  ordenar() {
    const idAtivo = this.participantes[this.ativoIndex]?.id;
    this.participantes.sort((a, b) => (Number(b.iniciativa) || 0) - (Number(a.iniciativa) || 0));

    if (idAtivo) {
      const novoIdx = this.participantes.findIndex(p => p.id === idAtivo);
      this.ativoIndex = novoIdx >= 0 ? novoIdx : 0;
    } else {
      this.ativoIndex = 0;
    }
  },

  adicionarParticipante(nome, iniciativa, fichaId = null) {
    const novo = {
      id: "part_" + Date.now() + Math.random().toString(36).substring(2, 5),
      nome: nome.trim(),
      iniciativa: Number(iniciativa) || 10,
      fichaId: fichaId || null,
      efeitos: []
    };
    this.participantes.push(novo);
    this.ordenar();
    this.sincronizarRede();
  },

  removerParticipante(id) {
    const idxRemover = this.participantes.findIndex(p => p.id === id);
    if (idxRemover < 0) return;

    // Preserva corretamente de quem é a vez mesmo se remover alguém antes dele na fila
    if (idxRemover < this.ativoIndex) {
      this.ativoIndex--;
    }

    this.participantes.splice(idxRemover, 1);

    if (this.ativoIndex >= this.participantes.length) {
      this.ativoIndex = 0;
    }
    this.sincronizarRede();
  },

  atualizarIniciativa(id, novaIniciativa) {
    const p = this.participantes.find(item => item.id === id);
    if (p) {
      p.iniciativa = Number(novaIniciativa) || 0;
      this.ordenar();
      this.sincronizarRede();
    }
  },

  avancarTurno() {
    if (!this.participantes.length) return;

    this.ativoIndex++;
    if (this.ativoIndex >= this.participantes.length) {
      this.ativoIndex = 0;
      this.rodada++;
    }

    // Ao iniciar o turno do participante, reduz em 1 os contadores de efeitos dele
    const atual = this.participantes[this.ativoIndex];
    if (atual && Array.isArray(atual.efeitos)) {
      atual.efeitos = atual.efeitos
        .map(ef => ({ ...ef, duracao: Number(ef.duracao) - 1 }))
        .filter(ef => ef.duracao > 0);
    }

    this.sincronizarRede();
  },

  adicionarEfeito(participanteId, nomeEfeito, duracaoTurnos) {
    const p = this.participantes.find(item => item.id === participanteId);
    if (p && nomeEfeito.trim()) {
      if (!Array.isArray(p.efeitos)) p.efeitos = [];
      p.efeitos.push({
        id: "ef_" + Date.now() + Math.random().toString(36).substring(2, 4),
        nome: nomeEfeito.trim(),
        duracao: Math.max(1, parseInt(duracaoTurnos, 10) || 1)
      });
      this.sincronizarRede();
    }
  },

  removerEfeito(participanteId, efeitoId) {
    const p = this.participantes.find(item => item.id === participanteId);
    if (p && Array.isArray(p.efeitos)) {
      p.efeitos = p.efeitos.filter(ef => ef.id !== efeitoId);
      this.sincronizarRede();
    }
  },

  limpar() {
    this.rodada = 1;
    this.ativoIndex = 0;
    this.participantes = [];
    this.sincronizarRede();
  }
};

export function renderizarCombate(container, store) {
  const badgeRodada = document.getElementById("round-badge");
  if (badgeRodada) {
    badgeRodada.textContent = `Turno ${store.rodada}`;
  }

  // Se o usuário estiver digitando num input de iniciativa, evita destruir o DOM sob os dedos dele
  const elAtivo = document.activeElement;
  if (elAtivo && elAtivo.classList.contains("init-input")) {
    return;
  }

  container.innerHTML = "";

  if (!store.participantes.length) {
    container.innerHTML = `
      <div style="text-align: center; color: var(--text-muted); padding: 28px 12px; font-size: 11px; line-height: 1.5;">
        Nenhum participante no combate.<br>Clique em <b>+ Participante</b> para iniciar.
      </div>
    `;
    return;
  }

  store.participantes.forEach((p, idx) => {
    const isCurrent = idx === store.ativoIndex;
    const itemCard = document.createElement("div");
    itemCard.className = `combat-item ${isCurrent ? "is-current" : ""}`;

    const efeitosHtml = (p.efeitos || []).map(ef => `
      <span class="effect-pill" data-pid="${p.id}" data-efid="${ef.id}" title="Clique para remover este efeito">
        ${ef.nome} (${ef.duracao}t)
        <span style="font-size:10px; margin-left:3px; opacity:0.8;">✕</span>
      </span>
    `).join("");

    itemCard.innerHTML = `
      <div class="combat-item-main">
        <span class="turn-dot" title="${isCurrent ? "Vez deste personagem" : ""}"></span>
        <span class="combat-item-name" title="${p.nome}">${p.nome}</span>
        <input type="number" class="init-input" data-id="${p.id}" value="${p.iniciativa}" title="Iniciativa (clique para alterar e tecle Enter)">
        <button type="button" class="btn-icon btn-del-part" data-id="${p.id}" title="Remover da lista">✕</button>
      </div>
      <div class="effects-row">
        ${efeitosHtml}
        <button type="button" class="btn-add-effect" data-id="${p.id}">+ Efeito</button>
      </div>
      <div class="effect-inline-form" id="ef-form-${p.id}">
        <input type="text" class="inp-ef-nome" placeholder="Ex: Envenenado...">
        <input type="number" class="inp-ef-turnos" value="3" min="1" title="Turnos">
        <button type="button" class="btn-ef-ok">OK</button>
      </div>
    `;

    // Edição manual de iniciativa (Salva no Enter ou ao sair do campo)
    const inpInit = itemCard.querySelector(".init-input");
    inpInit.addEventListener("keydown", (e) => {
      if (e.key === "Enter") inpInit.blur();
    });
    inpInit.addEventListener("blur", (e) => {
      const novoVal = Number(e.target.value);
      if (novoVal !== Number(p.iniciativa)) {
        store.atualizarIniciativa(p.id, novoVal);
      }
    });

    // Remover participante
    itemCard.querySelector(".btn-del-part").addEventListener("click", () => {
      store.removerParticipante(p.id);
    });

    // Abrir mini-form inline de efeito (sem usar window.prompt)
    const btnAddEf = itemCard.querySelector(".btn-add-effect");
    const formEf = itemCard.querySelector(`#ef-form-${p.id}`);
    const inpEfNome = formEf.querySelector(".inp-ef-nome");
    const inpEfTurnos = formEf.querySelector(".inp-ef-turnos");
    const btnEfOk = formEf.querySelector(".btn-ef-ok");

    btnAddEf.addEventListener("click", () => {
      const aberto = formEf.classList.toggle("is-open");
      if (aberto) inpEfNome.focus();
    });

    const confirmarEfeito = () => {
      const nome = inpEfNome.value.trim();
      const turnos = inpEfTurnos.value;
      if (nome) {

const STORAGE_KEYS = {
  BIN_ID: "solaria_cloud_bin_id",
  MASTER_KEY: "solaria_cloud_master_key",
  FICHAS: "solaria_fichas_custom",
  NOTAS: "solaria_notas_multiplas"

export const SolariaCloud = {
  obterCredenciais() {
    return {
      binId: (localStorage.getItem(STORAGE_KEYS.BIN_ID) || "").trim(),
      masterKey: (localStorage.getItem(STORAGE_KEYS.MASTER_KEY) || "").trim()
    };
  },

  salvarCredenciais(binId, masterKey) {
    localStorage.setItem(STORAGE_KEYS.BIN_ID, (binId || "").trim());
    localStorage.setItem(STORAGE_KEYS.MASTER_KEY, (masterKey || "").trim());
  },

  /**
   * Salva fichas e anotações no JSONBin.io
   */
  async salvarNuvem(binId, masterKey) {
    const bId = binId || this.obterCredenciais().binId;
    const mKey = masterKey || this.obterCredenciais().masterKey;

    if (!bId || !mKey) {
      throw new Error("Preencha o Bin ID e a Master Key antes de salvar.");
    }

    const fichas = JSON.parse(localStorage.getItem(STORAGE_KEYS.FICHAS) || "[]");
    const notas = JSON.parse(localStorage.getItem(STORAGE_KEYS.NOTAS) || "[]");

    const payload = {
      fichas,
      notas,
      atualizadoEm: new Date().toISOString()
    };

    const res = await fetch(`https://api.jsonbin.io/v3/b/${bId}`, {
      method: "PUT",
      headers: {
        "Content-Type": "application/json",
        "X-Master-Key": mKey
      },
      body: JSON.stringify(payload)
    });

    if (!res.ok) {
      const errData = await res.json().catch(() => ({}));
      throw new Error(errData.message || `Erro HTTP ${res.status} ao salvar na nuvem.`);
    }

    this.salvarCredenciais(bId, mKey);
    return { fichasQtd: fichas.length, notasQtd: notas.length };
  },

  /**
   * Baixa e restaura fichas e anotações do JSONBin.io
   */
  async baixarNuvem(binId, masterKey) {
    const bId = binId || this.obterCredenciais().binId;
    const mKey = masterKey || this.obterCredenciais().masterKey;

    if (!bId || !mKey) {
      throw new Error("Preencha o Bin ID e a Master Key antes de baixar.");
    }

    const res = await fetch(`https://api.jsonbin.io/v3/b/${bId}/latest`, {
      method: "GET",
      headers: {
        "X-Master-Key": mKey
      }
    });

    if (!res.ok) {
      const errData = await res.json().catch(() => ({}));
      throw new Error(errData.message || `Erro HTTP ${res.status} ao baixar da nuvem.`);
    }

    const data = await res.json();
    const record = data.record || {};

    const fichas = Array.isArray(record.fichas) ? record.fichas : [];
    const notas = Array.isArray(record.notas) ? record.notas : [];

    localStorage.setItem(STORAGE_KEYS.FICHAS, JSON.stringify(fichas));
    localStorage.setItem(STORAGE_KEYS.NOTAS, JSON.stringify(notas));
    this.salvarCredenciais(bId, mKey);

    // Se o usuário for Mestre, replica imediatamente para o Room Metadata e Broadcast
    try {
      const role = await OBR.player.getRole();
      if (role === "GM") {
        await SolariaSync.sincronizarFichasComSala(fichas);
        await SolariaSync.sincronizarNotasComSala(notas);
      }
      await OBR.broadcast.sendMessage("solaria/ficha-sync", fichas[0] || {});
      await OBR.broadcast.sendMessage("solaria/nota-sync", notas[0] || {});
    } catch (_) {}

    return { fichasQtd: fichas.length, notasQtd: notas.length };
  }
};

  width: 78,
  height: 24,
  radius: 8,
  spacing: 8
};

/**
 * Calcula dimensões escaladas e centro real do token no grid
 */
export function calcularGeometriaToken(token, sceneDpi = 150) {
  const scaleX = Math.abs(token.scale?.x ?? 1);
  const scaleY = Math.abs(token.scale?.y ?? 1);
  const imgW = token.image?.width || 150;
  const imgH = token.image?.height || 150;
  const tokenDpi = token.grid?.dpi || 150;

  const tokenWidth = (imgW / tokenDpi) * sceneDpi * scaleX;
  const tokenHeight = (imgH / tokenDpi) * sceneDpi * scaleY;

  const offsetX = token.grid?.offset?.x ?? (imgW / 2);
  const offsetY = token.grid?.offset?.y ?? (imgH / 2);

  const centerX = token.position.x + ((imgW / 2 - offsetX) / tokenDpi) * sceneDpi * scaleX;
  const centerY = token.position.y + ((imgH / 2 - offsetY) / tokenDpi) * sceneDpi * scaleY;

  const tokenRaio = Math.min(tokenWidth, tokenHeight) / 2;
  const tokenDiametro = tokenRaio * 2;
  const orbitalRadius = tokenRaio + BADGE_CONFIG.height / 2 + BADGE_CONFIG.spacing;

  return {
    centerX,
    centerY,
    tokenRaio,
    tokenDiametro,
    orbitalRadius
  };
}

/**
 * Calcula a posição x, y do badge orbital baseado no ângulo em graus
 */
export function calcularPosicaoOrbital(centerX, centerY, orbitalRadius, anguloGraus, badgeW, badgeH) {
  const rad = (anguloGraus * Math.PI) / 180;
  const px = centerX + orbitalRadius * Math.cos(rad);
  const py = centerY + orbitalRadius * Math.sin(rad);

  return {
    bx: Math.round(px - badgeW / 2),
    by: Math.round(py - badgeH / 2)
  };
}

/**
 * Formata os valores exibidos nos badges orbitais (atual/max)
 */
export function formatarValorStat(id, itemStat) {
  const cur = itemStat?.atual ?? 0;
  const max = itemStat?.max ?? cur;
  if (id === "pv" || id === "pm") return `${cur}/${max}`;
  if (cur === max || !max) return String(cur);
  return `${cur}/${max}`;
}

/**
 * Comandos vetoriais de curva para retângulos arredondados do Owlbear
 */
export function criarComandosRetanguloArredondado(w, h, r) {
  const raio = Math.min(r, w / 2, h / 2);
  return [
    [Command.MOVE, raio, 0],
    [Command.LINE, w - raio, 0],
    [Command.QUAD, w, 0, w, raio],
    [Command.LINE, w, h - raio],
    [Command.QUAD, w, h, w - raio, h],
    [Command.LINE, raio, h],
    [Command.QUAD, 0, h, 0, h - raio],
    [Command.LINE, 0, raio],
    [Command.QUAD, 0, 0, raio, 0],
    [Command.CLOSE]
  ];
}

export async function removerHudToken(tokenId, fichaId = null) {
  if (!tokenId && !fichaId) return;
  try {
    const role = await OBR.player.getRole();
    if (role !== "GM") return;

    const itensCena = await OBR.scene.items.getItems();
    const itensAntigos = itensCena.filter((item) => {
      const isHud = item.metadata?.[METADATA_HUD_KEY] || item.metadata?.[METADATA_BTN_FICHA_KEY];

      const matchFicha = fichaId && item.metadata?.sheetId === fichaId;
      return isHud && (matchToken || matchFicha);
    });

    }

/**
 * Ponto de entrada para desenhar ou atualizar os badges no mapa
 */
export async function atualizarHudToken(ficha, tokenIdEspecifico = null) {
  const tokenId = tokenIdEspecifico || ficha?.tokenId;
  if (!tokenId || !ficha) return;

  try {
    const role = await OBR.player.getRole();
    if (role !== "GM") return;
  } catch (_) {
    return;
  }

  const anterior = filaTokens.get(tokenId) || Promise.resolve();
  const proxima = anterior
    .catch(() => {})
    .then(() => _desenharHudInterno(ficha, tokenId))

      }
    });

  filaTokens.set(tokenId, proxima);
  return proxima;
}

async function _desenharHudInterno(ficha, tokenId) {
  try {
    await removerHudToken(tokenId, ficha.id);

    const items = await OBR.scene.items.getItems([tokenId]);
    const token = items[0];
    if (!token) return;

    let sceneDpi = 150;
    try { sceneDpi = await OBR.grid.getDpi(); } catch (_) {}

    const { centerX, centerY, tokenRaio, tokenDiametro, orbitalRadius } = calcularGeometriaToken(token, sceneDpi);
    const visivel = ficha.visivelPlayer !== false;
    const statsNorm = normalizarStats(ficha.stats);
    const novosItens = [];

    const statsAtivosGlobais = localStorage.getItem("solaria_exibir_stats") !== "false";

    if (statsAtivosGlobais) {
      const pvStat = statsNorm.pv || { atual: 0, max: 0 };
      const curPv = Number(pvStat.atual) || 0;
      const maxPv = Number(pvStat.max) || curPv;

      const isMorto = curPv <= 0;
      const isCritico = !isMorto && maxPv > 0 && curPv < (maxPv * 0.10);

      // 1. Círculo de Estado Crítico ou Derrotado
      if (isMorto || isCritico) {
        novosItens.push(construirAnelStatus({
          token,
          centerX,
          centerY,
          tokenDiametro,
          corAnel: isMorto ? "#e63946" : "#ff9e3b",
          visivel,
          fichaId: ficha.id,
          metadataKey: METADATA_HUD_KEY
        }));
      }

      // 2. Caveira Centralizada se Derrotado
      if (isMorto) {
        novosItens.push(construirCaveiraMorte({
          token,
          centerX,
          centerY,
          tokenDiametro,
          visivel,
          fichaId: ficha.id,
          metadataKey: METADATA_HUD_KEY
        }));

      const comandos = criarComandosRetanguloArredondado(BADGE_CONFIG.width, BADGE_CONFIG.height, BADGE_CONFIG.radius);

      ORBITA_STATS.forEach((cfg) => {
        const { bx, by } = calcularPosicaoOrbital(centerX, centerY, orbitalRadius, cfg.angulo, BADGE_CONFIG.width, BADGE_CONFIG.height);
        const textoValor = formatarValorStat(cfg.id, statsNorm[cfg.id]);

          token,
          bx,
          by,

          textoValor,
          visivel,
          fichaId: ficha.id,
          metadataKey: METADATA_HUD_KEY

    }));

/**
 * Avalia expressões matemáticas simples sem usar eval inseguro.
 * Suporta +, -, *, / e parênteses.
 */
function avaliarExpressaoMatematica(expr) {
  // Higieniza: permite apenas números e operadores matemáticos válidos
  const limpa = expr.replace(/[^0-9+\-*/().]/g, "");
  if (!limpa) return null;

  try {
    // Parser seguro via Function construtora isolada
    const resultado = new Function(`return (${limpa})`)();
    return typeof resultado === "number" && !isNaN(resultado) ? resultado : null;
  } catch (_) {
    return null;
  }
}

/**
 * Aplica o cálculo no atributo com base na entrada do usuário:
 * - Se começar com '+' ou '-': soma/subtrai do valor atual (ex: "-10", "+5", "-10/2")
 * - Se começar com '*' ou '/': multiplica/divide o valor atual (ex: "*2", "/2")
 * - Se for valor absoluto (ex: "50", "20+5"): define diretamente o resultado
 *
 * @param {number|string} valorAtual - O valor que o campo possui no momento
 * @param {string} inputDigitado - O que o usuário escreveu no input
 * @returns {number} O novo valor arredondado
 */
export function calcularNovoStat(valorAtual, inputDigitado) {
  const atual = parseFloat(valorAtual) || 0;
  const texto = String(inputDigitado).trim();

  if (!texto) return atual;

  const primeiroChar = texto.charAt(0);

  // Caso 1: Operações relativas ao valor atual (+, -, *, /)
  if (primeiroChar === "+" || primeiroChar === "-") {
    const delta = avaliarExpressaoMatematica(texto);
    if (delta !== null) {
      return Math.round(atual + delta);
    }
  } else if (primeiroChar === "*" || primeiroChar === "/") {
    const resultado = avaliarExpressaoMatematica(`${atual} ${texto}`);
    if (resultado !== null) {
      return Math.round(resultado);
    }
  }

  // Caso 2: Valor absoluto (ex: digitou apenas "15" ou uma conta direta "10+5")
  const valorDireto = avaliarExpressaoMatematica(texto);
  return valorDireto !== null ? Math.round(valorDireto) : atual;
}

import OBR from "https://esm.sh/@owlbear-rodeo/sdk@3.1.0";

export const METADATA_KEY = "solaria/sheetId";

export async function vincularTokenAoId(tokenId, sheetId) {
  if (!tokenId) return;
  try {
    await OBR.scene.items.updateItems([tokenId], (itens) => {
      for (const it of itens) it.metadata[METADATA_KEY] = sheetId;
    });
  } catch (_) {}
}

export async function desvincularToken(tokenId) {
  if (!tokenId) return;
  try {
    await OBR.scene.items.updateItems([tokenId], (itens) => {
      for (const it of itens) delete it.metadata[METADATA_KEY];
    });
  } catch (_) {}
}

export async function selecionarTokenNoGrid(tokenId) {
  if (tokenId) await OBR.player.select([tokenId]);
}

export async function obterDadosTokenSelecionado() {
  const selection = await OBR.player.getSelection();
  if (!selection || !selection.length) return null;

  const items = await OBR.scene.items.getItems([selection[0]]);
  const it = items[0];
  if (!it) return null;

  return {
    tokenId: it.id,
    tokenNome: it.text?.plainText || (it.name !== "Image" ? it.name : "") || "Token"
  };
}

export function iniciarOuvintesOwlbear(onFocarFicha, onCriarFicha) {
  // Foca a ficha quando clica no token no mapa
  OBR.player.onChange(async (player) => {
    if (!player.selection?.length) return;
    try {
      const items = await OBR.scene.items.getItems(player.selection);
      const sheetId = items[0]?.metadata?.[METADATA_KEY];
      if (sheetId) onFocarFicha(sheetId);
    } catch (_) {}
  });

  // Escuta clique do botão direito (menu de contexto)
  OBR.broadcast.onMessage("solaria/criar-ficha-token", (e) => {
    onCriarFicha(e.data.tokenId, e.data.tokenNome);
  });
}

/**
 * Normaliza o objeto de stats garantindo que cada atributo tenha { atual, max }.
 * Mantém retrocompatibilidade se existirem fichas antigas salvas com números simples.
 */
export function normalizarStats(statsRaw = {}) {
  const norm = {};
  STATS_CHAVES.forEach(({ id }) => {
    const s = statsRaw[id];
    if (typeof s === "object" && s !== null) {
      norm[id] = {
        atual: parseFloat(s.atual ?? s.max ?? 0),
        max: parseFloat(s.max ?? s.atual ?? 0)
      };
    } else {
      const val = parseFloat(s ?? 0) || 0;
      norm[id] = { atual: val, max: val };
    }
  });
  return norm;
}

const STORAGE_KEY = "solaria_fichas_custom";

export const SheetStore = {
  fichas: [],

  carregar(registrosEstaticos = []) {
    this.fichas = JSON.parse(localStorage.getItem(STORAGE_KEY) || "[]");

    registrosEstaticos.forEach(f => {
      if (!this.fichas.some(item => item.id === f.id)) {
        this.fichas.unshift({
          id: f.id,
          nome: f.nome,
          tag: f.tag || "NPC",
          visivelPlayer: f.visivelPlayer ?? true,
          stats: f.stats || { pv: "25", pm: "10", rm: "2", cd: "12", atk: "+4" },
          historico: f.desc || "",
          habilidades: f.habilidades || [],
          inventario: f.inventario || [],
          tokenId: null,
          tokenNome: null
        });
      }
    });
    return this.fichas;
  },

  salvar(notificarSala = true) {
    localStorage.setItem(STORAGE_KEY, JSON.stringify(this.fichas));
    if (notificarSala) {
      SolariaSync.sincronizarFichasComSala(this.fichas);
    }
  },

  obter(id) {
    return this.fichas.find(f => f.id === id);
  },

  adicionarOuAtualizar(dados, notificarSala = true) {
    const idx = this.fichas.findIndex(f => f.id === dados.id);
    if (idx >= 0) {
      this.fichas[idx] = dados;
    } else {
      this.fichas.unshift(dados);
    }
    this.salvar(notificarSala);
  },

  remover(id) {
    this.fichas = this.fichas.filter(f => f.id !== id);
    this.salvar(true);
  },

  /**
   * Mescla dados recebidos da sala (para clientes Jogador)
   */
  mesclarFichasPublicas(fichasPublicas = []) {
    this.fichas = fichasPublicas;
    localStorage.setItem(STORAGE_KEY, JSON.stringify(this.fichas));
  }
};

    const onPointerMove = (ev) => {
      let novoW = startW;
      let novoH = startH;

        novoH = startH + (startY - ev.screenY);
      }

      aplicarDimensoes(novoW, novoH);
    };

    const onPointerUp = () => {
      window.removeEventListener("pointermove", onPointerMove);
      window.removeEventListener("pointerup", onPointerUp);
    };

    window.addEventListener("pointermove", onPointerMove);
    window.addEventListener("pointerup", onPointerUp);
  }

function sincronizarModulosJson(rootDir = path.resolve(__dirname, "..")) {
  const painelDir = path.join(rootDir, "painel");
  const modulosPath = path.join(rootDir, "modulos.json");
  const estrutura = {};

  if (!fs.existsSync(painelDir)) return;

  const itens = fs.readdirSync(painelDir, { withFileTypes: true });

  itens.filter(i => i.isDirectory()).forEach(pasta => {
    const nomePasta = pasta.name;
    const caminhoPasta = path.join(painelDir, nomePasta);
    const arquivos = fs.readdirSync(caminhoPasta);

    estrutura[nomePasta] = {
      temCustomJS: arquivos.includes("module.js"),
      registros: []
    };

    arquivos.filter(f => f.endsWith(".json")).forEach(arq => {
      try {
        const conteudo = JSON.parse(fs.readFileSync(path.join(caminhoPasta, arq), "utf-8"));
        if (Array.isArray(conteudo)) {
          conteudo.forEach((item, idx) => {
            if (!item.id) item.id = `${nomePasta}-${arq.replace(".json", "")}-${idx}`;
            estrutura[nomePasta].registros.push(item);
          });
        } else if (typeof conteudo === "object" && conteudo !== null) {
          conteudo.id = conteudo.id || `${nomePasta}-${arq.replace(".json", "")}`;
          estrutura[nomePasta].registros.push(conteudo);
        }
      } catch (_) {}
    });
  });

  fs.writeFileSync(modulosPath, JSON.stringify(estrutura, null, 2), { encoding: "utf-8", flag: "w" });
}

function gerarProjetoCompleto(rootDir = path.resolve(__dirname, "..")) {
  if (isGenerating) return;
  isGenerating = true;

  try {
    const extensoesValidas = [".js", ".html", ".css", ".json"];
    const destino = path.join(rootDir, "projeto_completo.txt");
    let output = "";

    function varrerDiretorio(dir) {
      const itens = fs.readdirSync(dir, { withFileTypes: true });
      for (const item of itens) {
        const nome = item.name.toLowerCase();

        // Ignora estritamente pastas pesadas e qualquer variante do arquivo de snapshot
        if (
          nome === ".git" || 
          nome === "node_modules" || 
          nome.startsWith("projeto_completo") || 
          nome === "package-lock.json"
        ) {
          continue;
        }

        const fullPath = path.join(dir, item.name);
        if (item.isDirectory()) {
          varrerDiretorio(fullPath);
        } else if (item.isFile()) {
          const ext = path.extname(item.name).toLowerCase();
          if (extensoesValidas.includes(ext)) {
            const relPath = "\\" + path.relative(rootDir, fullPath);
            const content = fs.readFileSync(fullPath, "utf-8");
            output += `=== ARQUIVO: ${relPath} ===\n${content}\n\n`;
          }
        }
      }
    }

    varrerDiretorio(rootDir);

    // Limpa o arquivo primeiro e grava com flag 'w' explícita
    fs.writeFileSync(destino, output, { encoding: "utf-8", flag: "w" });
    console.log(`📝 [Builder] projeto_completo.txt regerado do ZERO com sucesso.`);
  } catch (err) {
    console.error("❌ [Builder] Erro ao gerar projeto_completo.txt:", err.message);
  } finally {
    isGenerating = false;
  }
}

async function aguardarDeployGitHubPages(versaoAlvo) {

  const maxTentativas = 48; // Limite de 4 minutos
  let tentativas = 0;
  const inicio = Date.now();

  const timer = setInterval(async () => {
    tentativas++;
    const decorridoSegundos = Math.round((Date.now() - inicio) / 1000);

    try {
      // Força bypass de cache no CDN do GitHub Pages com timestamp dinâmico
      const url = `${GITHUB_PAGES_MANIFEST}?_nocache=${Date.now()}`;

        }

        const json = await res.json().catch(() => ({}));
        
        if (json.version === versaoAlvo) {
          clearInterval(timer);
          try { process.stdout.write("\x07"); } catch (_) {} // Beep sonoro no terminal
          console.log(`\n==================================================`);
          console.log(`🎉 [GITHUB PAGES NO AR!] (levou ${decorridoSegundos}s)`);

function criarServidorHttp({ rootDir, port }) {
  return http.createServer((req, res) => {
    res.setHeader("Access-Control-Allow-Origin", "*");
    res.setHeader("Access-Control-Allow-Methods", "GET, OPTIONS, POST");
    res.setHeader("Access-Control-Allow-Headers", "*");
    res.setHeader("Access-Control-Allow-Private-Network", "true");
    res.setHeader("Cache-Control", "no-store, no-cache, must-revalidate, proxy-revalidate");
    res.setHeader("Pragma", "no-cache");
    res.setHeader("Expires", "0");

    if (req.method === "OPTIONS") {
      res.writeHead(204);
      return res.end();
    }

    const urlPath = req.url.split("?")[0];

    // Endpoint de Live Reload
    if (urlPath === "/livereload") {
      adicionarCliente(res);
      req.on("close", () => removerCliente(res));
      return;
    }

    let cleanPath = urlPath.replace(/^\/solaria-companion/, "");
    if (!cleanPath || cleanPath === "/") cleanPath = "/index.html";

    // Manifesto com sufixo (DEV Local)
    if (cleanPath === "/manifest.json") {
      const manifestPath = path.join(rootDir, "manifest.json");
      if (fs.existsSync(manifestPath)) {
        try {
          const manifest = JSON.parse(fs.readFileSync(manifestPath, "utf-8"));
          manifest.name = `${manifest.name || "Solaria"} (DEV Local)`;
          res.writeHead(200, { "Content-Type": "application/json; charset=utf-8" });
          return res.end(JSON.stringify(manifest, null, 2));
        } catch (_) {}
      }
    }

    const filePath = path.join(rootDir, cleanPath);

    if (fs.existsSync(filePath) && fs.statSync(filePath).isFile()) {
      const ext = path.extname(filePath).toLowerCase();
      const contentType = mimeTypes[ext] || "text/plain; charset=utf-8";

        let html = fs.readFileSync(filePath, "utf-8");

          html += SCRIPT_LIVE_RELOAD;

        res.writeHead(200, { "Content-Type": contentType });

      fs.createReadStream(filePath).pipe(res);
    } else {
      res.writeHead(404, { "Content-Type": "text/plain; charset=utf-8" });
      res.end("404 Not Found");
    }
  });
}

module.exports = { criarServidorHttp };

const fs = require("fs");
const { sincronizarModulosJson, gerarProjetoCompleto } = require("./builder");
const { notificarRecarregamento } = require("./livereload");

function iniciarWatcher(rootDir) {
  let watchTimeout = null;

  fs.watch(rootDir, { recursive: true }, (eventType, filename) => {
    if (!filename) return;

    const norm = filename.replace(/\\/g, "/").toLowerCase();

    // Ignora qualquer eco gerado pelos próprios arquivos de build
    if (
      norm.includes(".git/") || 
      norm.includes("node_modules/") || 
      norm.includes("modulos.json") || 
      norm.includes("projeto_completo")
    ) {
      return;
    }

    clearTimeout(watchTimeout);
    watchTimeout = setTimeout(() => {
      if (norm.startsWith("painel/")) {
        sincronizarModulosJson(rootDir);
      }
      gerarProjetoCompleto(rootDir);
      notificarRecarregamento();
    }, 250); // 250ms garante que o Windows terminou todas as rajadas de escrita
  });
}

module.exports = { iniciarWatcher };

/**
 * Cria programaticamente um nó de item de checklist via DOM puro
 * @param {string} textoInicial - Texto inicial do span
 * @returns {HTMLDivElement} Elemento .todo-item pronto
 */
export function criarElementoTodo(textoInicial = "") {
  const item = document.createElement("div");
  item.className = "todo-item";

  const check = document.createElement("input");
  check.type = "checkbox";
  check.className = "todo-check";
  // IMPORTANTE: contentEditable false impede que o cursor quebre a checkbox
  check.setAttribute("contenteditable", "false");

  const textSpan = document.createElement("span");
  textSpan.className = "todo-text";
  textSpan.textContent = textoInicial || "\u00a0"; // Espaço não-quebrável para manter foco

  item.appendChild(check);
  item.appendChild(textSpan);

  return item;
}

/**
 * Insere um novo item de tarefa na posição atual do cursor do editor
 * @param {HTMLDivElement} editorBody - Elemento com contenteditable
 * @param {Function} aoModificar - Callback para disparar debounce de salvamento
 */
export function inserirItemTodoNoCursor(editorBody, aoModificar) {
  editorBody.focus();

  const novoItem = criarElementoTodo("");
  const selection = window.getSelection();

  if (!selection || !selection.rangeCount) {
    editorBody.appendChild(novoItem);
  } else {
    const range = selection.getRangeAt(0);
    range.deleteContents();
    range.insertNode(novoItem);

    // Reposiciona o cursor dentro do span de texto da nova tarefa
    const spanTexto = novoItem.querySelector(".todo-text");
    if (spanTexto) {
      const novoRange = document.createRange();
      novoRange.selectNodeContents(spanTexto);
      novoRange.collapse(false);
      selection.removeAllRanges();
      selection.addRange(novoRange);
    }
  }

  if (typeof aoModificar === "function") aoModificar();
}

/**
 * Alterna a marcação visual e o atributo checked do elemento
 * @param {HTMLInputElement} checkbox - Elemento input checkbox clicado
 * @param {Function} aoModificar - Callback de salvamento
 */
export function alternarStatusCheck(checkbox, aoModificar) {
  const itemPai = checkbox.closest(".todo-item");
  if (!itemPai) return;

  if (checkbox.checked) {
    itemPai.classList.add("completed");
    checkbox.setAttribute("checked", "checked");
  } else {
    itemPai.classList.remove("completed");
    checkbox.removeAttribute("checked");
  }

  if (typeof aoModificar === "function") aoModificar();
}

/**
 * Garante que checkboxes preexistentes no HTML salvo mantenham contenteditable="false" e classes corretas
 * @param {HTMLDivElement} editorBody - Container do editor
 */
export function inicializarCheckboxesExistentes(editorBody) {
  editorBody.querySelectorAll(".todo-check").forEach((cb) => {
    cb.setAttribute("contenteditable", "false");
    const itemPai = cb.closest(".todo-item");
    if (cb.hasAttribute("checked") || itemPai?.classList.contains("completed")) {
      cb.checked = true;
      itemPai?.classList.add("completed");
    }
  });
}

 * Responsabilidade: Orquestrar toolbar, área editável e ciclo de vida do componente.
 */
export function criarRichEditor({ placeholder = "Digite seu texto...", valorInicial = "", onSalvar }) {
  // Injeta link da folha de estilos se ainda não presente no documento
  if (!document.getElementById("css-rich-editor-shared")) {
    const link = document.createElement("link");
    link.id = "css-rich-editor-shared";
    link.rel = "stylesheet";
    link.href = new URL("./rich-editor.css", import.meta.url).href;
    document.head.appendChild(link);
  }

  const container = document.createElement("div");
  container.className = "solaria-editor-container";

  const editorBody = document.createElement("div");
  editorBody.className = "solaria-editor-body";
  editorBody.setAttribute("contenteditable", "true");
  editorBody.setAttribute("data-placeholder", placeholder);
  editorBody.innerHTML = valorInicial || "";

  // Inicializa checkboxes persistidas
  inicializarCheckboxesExistentes(editorBody);

  // Debounce para persistência suave
  let timerSalvar = null;
  function notificarAlteracao() {
    clearTimeout(timerSalvar);
    timerSalvar = setTimeout(() => {
      if (typeof onSalvar === "function") {
        onSalvar(editorBody.innerHTML);
      }
    }, 400);
  }

  // Montagem da barra de ferramentas
  const toolbar = criarBarraFerramentas({
    aoFormatarTexto: (cmd) => {
      document.execCommand(cmd, false, null);
      notificarAlteracao();
    },
    aoInserirLista: () => {
      document.execCommand("insertUnorderedList", false, null);
      notificarAlteracao();
    },
    aoInserirTodo: () => {
      inserirItemTodoNoCursor(editorBody, notificarAlteracao);
    }
  });

  // Conexão de atalhos de teclado (Markdown e Enter)
  configurarAtalhosTeclado(editorBody, notificarAlteracao);

  // Eventos de clique e alteração nas checkboxes
  const tratarCliqueCheckbox = (e) => {
    if (e.target.classList.contains("todo-check")) {
      alternarStatusCheck(e.target, notificarAlteracao);
    }
  };

  editorBody.addEventListener("change", tratarCliqueCheckbox);
  editorBody.addEventListener("click", tratarCliqueCheckbox);
  editorBody.addEventListener("input", notificarAlteracao);

  // Acoplamento final
  container.appendChild(toolbar);
  container.appendChild(editorBody);

  return {
    elemento: container,
    obterConteudo: () => editorBody.innerHTML,
    definirConteudo: (html) => {
      editorBody.innerHTML = html || "";
      inicializarCheckboxesExistentes(editorBody);
    },
    focar: () => editorBody.focus()
  };
}

import { inserirItemTodoNoCursor } from "./checklist.js";

/**
 * Gerenciador de Atalhos de Teclado e Markdown
 * Responsabilidade: Tratar atalhos inline (* + Espaço, [] + Espaço) e quebras de linha com Enter.
 */

/**
 * Conecta os ouvintes de atalhos ao corpo do editor
 * @param {HTMLDivElement} editorBody - Elemento com contenteditable
 * @param {Function} aoModificar - Callback de salvamento
 */
export function configurarAtalhosTeclado(editorBody, aoModificar) {
  editorBody.addEventListener("keydown", (e) => {
    // Atalho 1: Markdown ao pressionar Espaço (* ou [])
    if (e.key === " " || e.code === "Space") {
      tratarEspacoMarkdown(e, editorBody, aoModificar);
      return;
    }

    // Atalho 2: Comportamento da tecla Enter dentro de checklists
    if (e.key === "Enter") {
      tratarEnterChecklist(e, editorBody, aoModificar);
    }
  });
}

function tratarEspacoMarkdown(e, editorBody, aoModificar) {
  const selection = window.getSelection();
  if (!selection || !selection.rangeCount) return;

  const range = selection.getRangeAt(0);
  const node = range.startContainer;

  if (node.nodeType !== Node.TEXT_NODE) return;

  const textoAnterior = node.textContent.slice(0, range.startOffset);
  const comando = textoAnterior.trim();

  // Lista com Marcadores (* ou -)
  if (comando === "*" || comando === "-") {
    e.preventDefault();
    const indiceInicio = textoAnterior.lastIndexOf(comando);
    const rangeRemocao = document.createRange();
    rangeRemocao.setStart(node, indiceInicio);
    rangeRemocao.setEnd(node, range.startOffset);

    selection.removeAllRanges();
    selection.addRange(rangeRemocao);
    document.execCommand("delete");
    document.execCommand("insertUnorderedList");

    if (typeof aoModificar === "function") aoModificar();
    return;
  }

  // Checklist ([] ou [ ])
  if (comando === "[]" || comando === "[ ]") {
    e.preventDefault();
    const indiceInicio = textoAnterior.lastIndexOf(comando);
    const rangeRemocao = document.createRange();
    rangeRemocao.setStart(node, indiceInicio);
    rangeRemocao.setEnd(node, range.startOffset);

    selection.removeAllRanges();
    selection.addRange(rangeRemocao);
    document.execCommand("delete");

    inserirItemTodoNoCursor(editorBody, aoModificar);
  }
}

function tratarEnterChecklist(e, editorBody, aoModificar) {
  const selection = window.getSelection();
  if (!selection || !selection.rangeCount) return;

  const range = selection.getRangeAt(0);
  const elementoAlvo = range.startContainer.nodeType === Node.ELEMENT_NODE
    ? range.startContainer
    : range.startContainer.parentElement;

  const todoAtual = elementoAlvo?.closest(".todo-item");
  if (!todoAtual) return;

  const spanTexto = todoAtual.querySelector(".todo-text");
  const conteudoTexto = spanTexto ? spanTexto.textContent.replace(/\u00a0/g, "").trim() : "";

  // Se deu Enter em item vazio, encerra a lista de tarefas e abre parágrafo comum
  if (!conteudoTexto) {
    e.preventDefault();
    todoAtual.remove();
    document.execCommand("insertParagraph");
    if (typeof aoModificar === "function") aoModificar();
    return;
  }

  // Se tem texto, quebra criando outro item de checklist logo abaixo
  e.preventDefault();
  inserirItemTodoNoCursor(editorBody, aoModificar);
}

/**
 * Monta o nó da barra de ferramentas com botões configurados
 * @param {Object} callbacks - Funções chamadas pelas ações especiais
 * @param {Function} callbacks.aoFormatarTexto - Executa document.execCommand
 * @param {Function} callbacks.aoInserirLista - Insere lista com marcadores
 * @param {Function} callbacks.aoInserirTodo - Insere nó de checklist
 * @returns {HTMLDivElement} Elemento .solaria-editor-toolbar
 */
export function criarBarraFerramentas({ aoFormatarTexto, aoInserirLista, aoInserirTodo }) {

  toolbar.className = "solaria-editor-toolbar";

  // 1. Botões de comando direto (B, I, U, S)
  ACOES_FORMATACAO.forEach(({ rotulo, tagInterna, cmd, dica }) => {
    const btn = document.createElement("button");
    btn.type = "button";
    btn.className = "ed-btn";
    btn.title = dica;

    const rotuloEl = document.createElement(tagInterna);
    rotuloEl.textContent = rotulo;
    btn.appendChild(rotuloEl);

    btn.addEventListener("mousedown", (e) => {
      e.preventDefault(); // Impede perda de seleção no editor
      if (typeof aoFormatarTexto === "function") aoFormatarTexto(cmd);
    });

    toolbar.appendChild(btn);
  });

  // Divisor visual
  const divisor = document.createElement("div");
  divisor.className = "ed-divider";
  toolbar.appendChild(divisor);

  // 2. Botão Lista com Marcadores
  const btnLista = document.createElement("button");
  btnLista.type = "button";
  btnLista.className = "ed-btn";
  btnLista.textContent = "• Lista";
  btnLista.title = "Lista com Marcadores (* + Espaço)";
  btnLista.addEventListener("mousedown", (e) => {
    e.preventDefault();
    if (typeof aoInserirLista === "function") aoInserirLista();
  });
  toolbar.appendChild(btnLista);
