import OBR from "https://esm.sh/@owlbear-rodeo/sdk@3.1.0";
import { SolariaSync } from "./sync.js";

const STORAGE_KEYS = {
  BIN_ID: "solaria_cloud_bin_id",
  MASTER_KEY: "solaria_cloud_master_key",
  FICHAS: "solaria_fichas_custom",
  NOTAS: "solaria_notas_multiplas"
};

export const SolariaCloud = {
  obterCredenciais() {
    return {
      binId: (localStorage.getItem(STORAGE_KEYS.BIN_ID) || "").trim(),
      masterKey: (localStorage.getItem(STORAGE_KEYS.MASTER_KEY) || "").trim()
    };
  },

  salvarCredenciais(binId, masterKey) {
    localStorage.setItem(STORAGE_KEYS.BIN_ID, (binId || "").trim());
    localStorage.setItem(STORAGE_KEYS.MASTER_KEY, (masterKey || "").trim());
  },

  /**
   * Salva fichas e anotações no JSONBin.io
   */
  async salvarNuvem(binId, masterKey) {
    const bId = binId || this.obterCredenciais().binId;
    const mKey = masterKey || this.obterCredenciais().masterKey;

    if (!bId || !mKey) {
      throw new Error("Preencha o Bin ID e a Master Key antes de salvar.");
    }

    const fichas = JSON.parse(localStorage.getItem(STORAGE_KEYS.FICHAS) || "[]");
    const notas = JSON.parse(localStorage.getItem(STORAGE_KEYS.NOTAS) || "[]");

    const payload = {
      fichas,
      notas,
      atualizadoEm: new Date().toISOString()
    };

    const res = await fetch(`https://api.jsonbin.io/v3/b/${bId}`, {
      method: "PUT",
      headers: {
        "Content-Type": "application/json",
        "X-Master-Key": mKey
      },
      body: JSON.stringify(payload)
    });

    if (!res.ok) {
      const errData = await res.json().catch(() => ({}));
      throw new Error(errData.message || `Erro HTTP ${res.status} ao salvar na nuvem.`);
    }

    this.salvarCredenciais(bId, mKey);
    return { fichasQtd: fichas.length, notasQtd: notas.length };
  },

  /**
   * Baixa e restaura fichas e anotações do JSONBin.io
   */
  async baixarNuvem(binId, masterKey) {
    const bId = binId || this.obterCredenciais().binId;
    const mKey = masterKey || this.obterCredenciais().masterKey;

    if (!bId || !mKey) {
      throw new Error("Preencha o Bin ID e a Master Key antes de baixar.");
    }

    const res = await fetch(`https://api.jsonbin.io/v3/b/${bId}/latest`, {
      method: "GET",
      headers: {
        "X-Master-Key": mKey
      }
    });

    if (!res.ok) {
      const errData = await res.json().catch(() => ({}));
      throw new Error(errData.message || `Erro HTTP ${res.status} ao baixar da nuvem.`);
    }

    const data = await res.json();
    const record = data.record || {};

    const fichas = Array.isArray(record.fichas) ? record.fichas : [];
    const notas = Array.isArray(record.notas) ? record.notas : [];

    localStorage.setItem(STORAGE_KEYS.FICHAS, JSON.stringify(fichas));
    localStorage.setItem(STORAGE_KEYS.NOTAS, JSON.stringify(notas));
    this.salvarCredenciais(bId, mKey);

    // Se o usuário for Mestre, replica imediatamente para o Room Metadata e Broadcast
    try {
      const role = await OBR.player.getRole();
      if (role === "GM") {
        await SolariaSync.sincronizarFichasComSala(fichas);
        await SolariaSync.sincronizarNotasComSala(notas);
      }
      await OBR.broadcast.sendMessage("solaria/ficha-sync", fichas[0] || {});
      await OBR.broadcast.sendMessage("solaria/nota-sync", notas[0] || {});
    } catch (_) {}

    return { fichasQtd: fichas.length, notasQtd: notas.length };
  }
};

export function configurarInterfaceNuvem() {
  const inpBinId = document.getElementById("cloud-bin-id");
  const inpKey = document.getElementById("cloud-master-key");
  const btnSalvar = document.getElementById("btn-salvar-nuvem");
  const btnBaixar = document.getElementById("btn-baixar-nuvem");

  if (!inpBinId || !inpKey || !btnSalvar || !btnBaixar) return;

  const creds = SolariaCloud.obterCredenciais();
  inpBinId.value = creds.binId;
  inpKey.value = creds.masterKey;

  const salvarInputs = () => {
    SolariaCloud.salvarCredenciais(inpBinId.value, inpKey.value);
  };

  inpBinId.addEventListener("input", salvarInputs);
  inpKey.addEventListener("input", salvarInputs);

  function exibirStatus(msg, cor = "#7982a9") {
    if (statusEl) {
      statusEl.textContent = msg;
      statusEl.style.color = cor;
    }
  }

  btnSalvar.addEventListener("click", async () => {

      } catch (_) {}
    } catch (err) {
      exibirStatus(err.message, "#f7768e");
    } finally {
      btnSalvar.disabled = false;
    }
  });

  btnBaixar.addEventListener("click", async () => {
    try {
      exibirStatus("Baixando da nuvem...", "#e0af68");

      btnBaixar.disabled = false;

  });

export const SYNC_KEYS = {
  FICHAS: "solaria/sync_fichas",
  NOTAS: "solaria/sync_notas"
};

export const SolariaSync = {
  /**
   * Mestre publica todas as fichas visíveis nos metadados da sala
   */
  async sincronizarFichasComSala(fichas = []) {
    try {
      const role = await OBR.player.getRole();
      if (role !== "GM") return;

      const fichasPublicas = fichas.filter(f => f.visivelPlayer !== false);
      await OBR.room.setMetadata({
        [SYNC_KEYS.FICHAS]: fichasPublicas
      });
    } catch (err) {
      console.error("Erro ao sincronizar fichas no Room Metadata:", err);
    }
  },

  /**
   * Mestre publica todas as anotações visíveis nos metadados da sala
   */
  async sincronizarNotasComSala(notas = []) {
    try {
      const role = await OBR.player.getRole();
      if (role !== "GM") return;

      const notasPublicas = notas.filter(n => n.visivelPlayer === true);
      await OBR.room.setMetadata({
        [SYNC_KEYS.NOTAS]: notasPublicas
      });
    } catch (err) {
      console.error("Erro ao sincronizar notas no Room Metadata:", err);
    }
  },

  /**
   * Inicia escuta contínua de Room Metadata para atualizar o cliente do Jogador
   */
  iniciarOuvinteSala({ onFichasRecebidas, onNotasRecebidas }) {
    // 1. Carrega dados iniciais da sala imediatamente
    OBR.room.getMetadata().then((meta) => {
      if (meta?.[SYNC_KEYS.FICHAS] && typeof onFichasRecebidas === "function") {
        onFichasRecebidas(meta[SYNC_KEYS.FICHAS]);
      }
      if (meta?.[SYNC_KEYS.NOTAS] && typeof onNotasRecebidas === "function") {
        onNotasRecebidas(meta[SYNC_KEYS.NOTAS]);
      }
    }).catch(() => {});

    // 2. Ouve alterações em tempo real na sala
    return OBR.room.onMetadataChange((meta) => {
      if (meta?.[SYNC_KEYS.FICHAS] && typeof onFichasRecebidas === "function") {
        onFichasRecebidas(meta[SYNC_KEYS.FICHAS]);
      }
      if (meta?.[SYNC_KEYS.NOTAS] && typeof onNotasRecebidas === "function") {
        onNotasRecebidas(meta[SYNC_KEYS.NOTAS]);
      }
    });
  }
};

export async function abrirOuFocarPainel() {
  if (!aberto) {
    const salvo = JSON.parse(localStorage.getItem("solaria_painel_dimensoes") || "{}");
    const width = salvo.width || 480;
    const height = salvo.height || 680;

    await OBR.popover.open({
      id: ID_PAINEL,
      url: BASE_URL + "/painel/painel.html",
      height: height,
      width: width,
      anchorOrigin: { horizontal: "RIGHT", vertical: "BOTTOM" },
      transformOrigin: { horizontal: "RIGHT", vertical: "BOTTOM" },
      disableClickAway: true
    });
    aberto = true;
  }

async function inicializarHudsAoCarregarCena() {
  try {
    const res = await fetch(BASE_URL + "/modulos.json");
    const modulos = await res.json();
    let fichas = JSON.parse(localStorage.getItem("solaria_fichas_custom") || "[]");

    if (modulos.fichas?.registros) {
      modulos.fichas.registros.forEach(f => {
        if (!fichas.some(item => item.id === f.id)) {
          fichas.push(f);
        }
      });
    }

    await sincronizarTodosHuds(fichas);
  } catch (err) {
    console.error("Erro ao sincronizar HUDs iniciais:", err);
  }
}

export const criarFichaOption = {
  id: "solaria/context-menu-criar-ficha",
  roleRequired: "GM",
  config: {
    icons: [
      {
        icon: "/favicon.svg",
        label: "Criar Ficha",
        filter: {
          every: [
            { key: "layer", value: "CHARACTER", coordinator: "||" },
            { key: "layer", value: "PROP", coordinator: "||" },
            { key: "type", value: "IMAGE" }
          ],
          permissions: ["UPDATE"]
        }
      }
    ]
  },
  async onClick(context, helpers) {
    const item = context.items[0];
    if (!item) return;

    const tokenId = item.id;
    // Prioriza o nome ou rótulo de texto atribuído ao token no mapa
    const tokenNome = item.text?.plainText || (item.name && item.name !== "Image" ? item.name : "") || item.name || "Novo Token";

    // 1. Salva a intenção no localStorage para não perder a ação se o painel estiver abrindo
    localStorage.setItem("solaria_acao_pendente", JSON.stringify({
      tipo: "CRIAR_FICHA",
      tokenId,
      tokenNome,
      timestamp: Date.now()
    }));

    // 2. Garante que o painel seja aberto
    if (helpers?.abrirOuFocarPainel) {
      await helpers.abrirOuFocarPainel();
    }

    // 3. Emite broadcast imediato caso o painel já esteja carregado e aberto na tela
    try {
      await OBR.broadcast.sendMessage("solaria/criar-ficha-token", {
        tokenId,
        tokenNome
      });
    } catch (_) {}
  }
};

  for (const item of listaOpcoes) {

    if (item.roleRequired && item.roleRequired !== currentRole) {
      continue;
    }

    OBR.contextMenu.create({
      id: item.id,
      icons: item.config.icons,
      onClick: async (context) => {
        await item.onClick(context, helpers);
      }
    });
  }

  function popularOpcoes() {
    select.innerHTML = "";

    const fichasCustom = JSON.parse(localStorage.getItem("solaria_fichas_custom") || "[]");
    fichasCustom
      .filter(f => Estado.role === "GM" || f.visivelPlayer)
      .forEach(f => {
        const opt = document.createElement("option");
        opt.value = JSON.stringify({ id: f.id, nome: f.nome, aba: "fichas" });
        opt.textContent = `[FICHA] ${f.nome}`;
        select.appendChild(opt);
      });

    Object.entries(Estado.modulos || {}).forEach(([aba, dados]) => {
      if (aba === "anotacoes" || aba === "fichas") return;
      (dados.registros || [])
        .filter(i => Estado.role === "GM" || i.visivelPlayer)
        .forEach(item => {
          const opt = document.createElement("option");
          opt.value = JSON.stringify({ id: item.id, nome: item.nome, aba });
          opt.textContent = `[${aba.toUpperCase()}] ${item.nome}`;
          select.appendChild(opt);
        });
    });
  }

  function renderizarChips() {
    chipsContainer.innerHTML = "";
    if (!anexos.length) {
      chipsContainer.innerHTML = `<span style="font-size:11px; color:#565f89; padding:2px;">Nenhum atalho fixado.</span>`;
      return;
    }

    anexos.forEach((anexo, idx) => {
      const chip = document.createElement("div");
      chip.className = "chip-badge";

      let catClass = "cat-default";
      if (anexo.aba === "fichas") catClass = "cat-fichas";

      });

import { SolariaSync } from "../../core/sync.js";

const STORAGE_KEY = "solaria_notas_multiplas";

export const NotesStore = {
  notas: [],
  idAtiva: null,

  carregar() {
    try {
      const salvas = JSON.parse(localStorage.getItem(STORAGE_KEY) || "null");
      if (Array.isArray(salvas)) {
        this.notas = salvas;
      } else {
        this.notas = [];
      }
    } catch (_) {
      this.notas = [];
    }

    if (!this.idAtiva || !this.notas.some(n => String(n.id) === String(this.idAtiva))) {
      this.idAtiva = this.notas[0]?.id || null;
    }
    return this.notas;
  },

  salvar(notificarSala = true) {
    localStorage.setItem(STORAGE_KEY, JSON.stringify(this.notas));
    if (notificarSala) {
      SolariaSync.sincronizarNotasComSala(this.notas);
    }
  },

  obterAtiva() {
    return this.notas.find(n => String(n.id) === String(this.idAtiva)) || null;
  },

  definirAtiva(id) {
    this.idAtiva = id;
  },

  criarNovaNota(titulo = "Nova Anotação") {
    const nova = {
      id: "note_" + Date.now(),
      titulo,
      conteudo: "",
      visivelPlayer: false,
      atualizadaEm: Date.now()
    };
    this.notas.unshift(nova);
    this.idAtiva = nova.id;
    this.salvar(true);
    return nova;
  },

  atualizarNota(id, dados) {
    const nota = this.notas.find(n => String(n.id) === String(id));
    if (!nota) return;
    Object.assign(nota, dados, { atualizadaEm: Date.now() });
    this.salvar(true);
  },

  excluirNota(id) {
    this.notas = this.notas.filter(n => String(n.id) !== String(id));
    if (String(this.idAtiva) === String(id)) {
      this.idAtiva = this.notas[0]?.id || null;
    }
    this.salvar(true);
  },

  /**
   * Mescla notas públicas recebidas do Room Metadata
   */
  mesclarNotasPublicas(notasPublicas = []) {
    this.notas = notasPublicas;
    localStorage.setItem(STORAGE_KEY, JSON.stringify(this.notas));
  }
};

/**
 * Avalia expressões matemáticas simples sem usar eval inseguro.
 * Suporta +, -, *, / e parênteses.
 */
function avaliarExpressaoMatematica(expr) {
  // Higieniza: permite apenas números e operadores matemáticos válidos
  const limpa = expr.replace(/[^0-9+\-*/().]/g, "");
  if (!limpa) return null;

  try {
    // Parser seguro via Function construtora isolada
    const resultado = new Function(`return (${limpa})`)();
    return typeof resultado === "number" && !isNaN(resultado) ? resultado : null;
  } catch (_) {
    return null;
  }
}

/**
 * Aplica o cálculo no atributo com base na entrada do usuário:
 * - Se começar com '+' ou '-': soma/subtrai do valor atual (ex: "-10", "+5", "-10/2")
 * - Se começar com '*' ou '/': multiplica/divide o valor atual (ex: "*2", "/2")
 * - Se for valor absoluto (ex: "50", "20+5"): define diretamente o resultado
 *
 * @param {number|string} valorAtual - O valor que o campo possui no momento
 * @param {string} inputDigitado - O que o usuário escreveu no input
 * @returns {number} O novo valor arredondado
 */
export function calcularNovoStat(valorAtual, inputDigitado) {
  const atual = parseFloat(valorAtual) || 0;
  const texto = String(inputDigitado).trim();

  if (!texto) return atual;

  const primeiroChar = texto.charAt(0);

  // Caso 1: Operações relativas ao valor atual (+, -, *, /)
  if (primeiroChar === "+" || primeiroChar === "-") {
    const delta = avaliarExpressaoMatematica(texto);
    if (delta !== null) {
      return Math.round(atual + delta);
    }
  } else if (primeiroChar === "*" || primeiroChar === "/") {
    const resultado = avaliarExpressaoMatematica(`${atual} ${texto}`);
    if (resultado !== null) {
      return Math.round(resultado);
    }
  }

  // Caso 2: Valor absoluto (ex: digitou apenas "15" ou uma conta direta "10+5")
  const valorDireto = avaliarExpressaoMatematica(texto);
  return valorDireto !== null ? Math.round(valorDireto) : atual;
}

export const METADATA_KEY = "solaria/sheetId";

export async function vincularTokenAoId(tokenId, sheetId) {
  if (!tokenId) return;
  try {
    await OBR.scene.items.updateItems([tokenId], (itens) => {
      for (const it of itens) it.metadata[METADATA_KEY] = sheetId;
    });
  } catch (_) {}
}

export async function desvincularToken(tokenId) {
  if (!tokenId) return;
  try {
    await OBR.scene.items.updateItems([tokenId], (itens) => {
      for (const it of itens) delete it.metadata[METADATA_KEY];
    });
  } catch (_) {}
}

export async function selecionarTokenNoGrid(tokenId) {
  if (tokenId) await OBR.player.select([tokenId]);
}

export async function obterDadosTokenSelecionado() {
  const selection = await OBR.player.getSelection();
  if (!selection || !selection.length) return null;

  const items = await OBR.scene.items.getItems([selection[0]]);
  const it = items[0];
  if (!it) return null;

  return {
    tokenId: it.id,
    tokenNome: it.text?.plainText || (it.name !== "Image" ? it.name : "") || "Token"
  };
}

export function iniciarOuvintesOwlbear(onFocarFicha, onCriarFicha) {

      const sheetId = items[0]?.metadata?.[METADATA_KEY];
      if (sheetId) onFocarFicha(sheetId);
    } catch (_) {}
  });

  // Escuta clique do botão direito (menu de contexto)
  OBR.broadcast.onMessage("solaria/criar-ficha-token", (e) => {

/**
 * Normaliza o objeto de stats garantindo que cada atributo tenha { atual, max }.
 * Mantém retrocompatibilidade se existirem fichas antigas salvas com números simples.
 */
export function normalizarStats(statsRaw = {}) {
  const norm = {};
  STATS_CHAVES.forEach(({ id }) => {
    const s = statsRaw[id];
    if (typeof s === "object" && s !== null) {
      norm[id] = {
        atual: parseFloat(s.atual ?? s.max ?? 0),
        max: parseFloat(s.max ?? s.atual ?? 0)
      };
    } else {
      const val = parseFloat(s ?? 0) || 0;
      norm[id] = { atual: val, max: val };
    }
  });
  return norm;
}

function formatarValorStat(id, itemStat) {
  const cur = itemStat?.atual ?? 0;
  const max = itemStat?.max ?? cur;
  if (id === "pv" || id === "pm") return `${cur}/${max}`;
  if (cur === max || !max) return String(cur);
  return `${cur}/${max}`;
}

function criarComandosRetanguloArredondado(w, h, r) {
  const raio = Math.min(r, w / 2, h / 2);
  return [
    [Command.MOVE, raio, 0],
    [Command.LINE, w - raio, 0],
    [Command.QUAD, w, 0, w, raio],
    [Command.LINE, w, h - raio],
    [Command.QUAD, w, h, w - raio, h],
    [Command.LINE, raio, h],
    [Command.QUAD, 0, h, 0, h - raio],
    [Command.LINE, 0, raio],
    [Command.QUAD, 0, 0, raio, 0],
    [Command.CLOSE]
  ];
}

export async function removerHudToken(tokenId, fichaId = null) {
  if (!tokenId && !fichaId) return;
  try {
    const itensCena = await OBR.scene.items.getItems();
    const itensAntigos = itensCena.filter((item) => {
      const isHud = item.metadata?.[METADATA_HUD_KEY] || item.metadata?.[METADATA_BTN_FICHA_KEY];
      const matchToken = tokenId && item.attachedTo === tokenId;
      const matchFicha = fichaId && item.metadata?.sheetId === fichaId;
      return isHud && (matchToken || matchFicha);
    });

    if (itensAntigos.length > 0) {
      await OBR.scene.items.deleteItems(itensAntigos.map((i) => i.id));
    }
  } catch (err) {
    console.error("Erro ao remover HUD:", err);
  }
}

function configurarArrasto() {
  const elTop = document.getElementById("resize-top");
  const elLeft = document.getElementById("resize-left");
  const elCorner = document.getElementById("resize-corner");

  function iniciarRedimensionamento(e, redimensionaW, redimensionaH) {
    e.preventDefault();
    const startX = e.screenX;
    const startY = e.screenY;
    const startW = Estado.dimensoes.width;
    const startH = Estado.dimensoes.height;

    const onPointerMove = (ev) => {
      let novoW = startW;
      let novoH = startH;

      if (redimensionaW) {
        novoW = startW + (startX - ev.screenX);
      }
      if (redimensionaH) {
        novoH = startH + (startY - ev.screenY);
      }

      aplicarDimensoes(novoW, novoH);
    };

    const onPointerUp = () => {
      window.removeEventListener("pointermove", onPointerMove);
      window.removeEventListener("pointerup", onPointerUp);
    };

    window.addEventListener("pointermove", onPointerMove);
    window.addEventListener("pointerup", onPointerUp);
  }

  elTop?.addEventListener("pointerdown", (e) => iniciarRedimensionamento(e, false, true));
  elLeft?.addEventListener("pointerdown", (e) => iniciarRedimensionamento(e, true, false));
  elCorner?.addEventListener("pointerdown", (e) => iniciarRedimensionamento(e, true, true));

  const btnPreset = document.getElementById("btn-toggle-size");

    { label: "📐 Médio", width: 480, height: 680 },

}

function sincronizarModulosJson() {

  const modulosPath = path.join(__dirname, "modulos.json");
  const estrutura = {};

  if (!fs.existsSync(painelDir)) return;

    const nomePasta = pasta.name;

      try {
        const conteudo = JSON.parse(fs.readFileSync(path.join(caminhoPasta, arq), "utf-8"));
        if (Array.isArray(conteudo)) {
          estrutura[nomePasta].registros.push(...conteudo);
        } else {
          conteudo.id = conteudo.id || `${nomePasta}-${arq.replace(".json", "")}`;
          estrutura[nomePasta].registros.push(conteudo);
        }
      } catch (_) {}
    });
  });

}

        color: #c0caf5;
        font-size: 13px;
        line-height: 1.6;
        outline: none;
        overflow-y: auto;
      }
      .solaria-editor-body[contenteditable]:empty::before {
        content: attr(data-placeholder);
        color: #565f89;
        pointer-events: none;
      }

      }

      }
      .todo-check {
        cursor: pointer;
        margin-top: 3px;
        accent-color: #7aa2f7;
      }
      .todo-text {
        flex: 1;

      .todo-item.completed .todo-text {
        text-decoration: line-through;
        color: #565f89 !important;
      }
    `;
    document.head.appendChild(style);
  }

  const container = document.createElement("div");
  container.className = "solaria-editor-container";

  container.innerHTML = `
    <div class="solaria-editor-toolbar">
      <button type="button" class="ed-btn" data-cmd="bold" title="Negrito (Ctrl+B)"><b>B</b></button>
      <button type="button" class="ed-btn" data-cmd="italic" title="Itálico (Ctrl+I)"><i>I</i></button>
      <button type="button" class="ed-btn" data-cmd="underline" title="Sublinhado (Ctrl+U)"><u>U</u></button>
      <button type="button" class="ed-btn" data-cmd="strikeThrough" title="Tachado"><s>S</s></button>
      <div class="ed-divider"></div>
      <button type="button" class="ed-btn" data-action="list" title="Lista com Marcadores (* + Espaço)">• Lista</button>
      <button type="button" class="ed-btn" data-action="todo" title="Checklist ([] + Espaço)">☑ Tarefa</button>
    </div>
    <div class="solaria-editor-body" contenteditable="true" data-placeholder="${placeholder}"></div>
  `;

  const body = container.querySelector(".solaria-editor-body");
  body.innerHTML = valorInicial || "";

  // Sincroniza estado de checkboxes carregados do HTML salvo
  body.querySelectorAll(".todo-check").forEach((cb) => {
    cb.setAttribute("contenteditable", "false");
    if (cb.hasAttribute("checked") || cb.closest(".todo-item")?.classList.contains("completed")) {
      cb.checked = true;
      cb.closest(".todo-item")?.classList.add("completed");
    }
  });

  let debounceTimer;
  function notificarAlteracao() {
    clearTimeout(debounceTimer);
    debounceTimer = setTimeout(() => {
      if (typeof onSalvar === "function") {
        onSalvar(body.innerHTML);
      }
    }, 400);
  }

  function inserirItemTodo() {
    body.focus();
    const tempId = "todo_" + Date.now();
    const todoHtml = `<div class="todo-item" id="${tempId}"><input type="checkbox" class="todo-check" contenteditable="false"><span class="todo-text">&nbsp;</span></div>`;
    document.execCommand("insertHTML", false, todoHtml);

    const el = body.querySelector(`#${tempId}`);
    if (el) {
      el.removeAttribute("id");
      const span = el.querySelector(".todo-text");
      if (span) {
        const sel = window.getSelection();
        const r = document.createRange();
        r.selectNodeContents(span);
        r.collapse(false);
        sel.removeAllRanges();
        sel.addRange(r);
      }
    }
    notificarAlteracao();
  }

  // Botões de Formatação Simples
  container.querySelectorAll(".ed-btn[data-cmd]").forEach((btn) => {
    btn.addEventListener("mousedown", (e) => {
      e.preventDefault();
      document.execCommand(btn.dataset.cmd, false, null);
      notificarAlteracao();
    });
  });

  // Botões da Barra
  container.querySelector('[data-action="list"]').addEventListener("mousedown", (e) => {
    e.preventDefault();
    document.execCommand("insertUnorderedList", false, null);
    notificarAlteracao();
  });

  container.querySelector('[data-action="todo"]').addEventListener("mousedown", (e) => {
    e.preventDefault();
    inserirItemTodo();
  });

  // Interceptação de Atalhos no Teclado
  body.addEventListener("keydown", (e) => {
    // 1. Atalhos Markdown ao pressionar Espaço (* + Espaço ou [] + Espaço)
    if (e.key === " " || e.code === "Space") {
      const selection = window.getSelection();
      if (!selection || !selection.rangeCount) return;
      const range = selection.getRangeAt(0);
      const node = range.startContainer;

      if (node.nodeType === Node.TEXT_NODE) {
        const textBefore = node.textContent.slice(0, range.startOffset);
        const trimmed = textBefore.trim();

        // Lista (* ou -)
        if (trimmed === "*" || trimmed === "-") {
          e.preventDefault();
          const startIdx = textBefore.lastIndexOf(trimmed);
          const delRange = document.createRange();
          delRange.setStart(node, startIdx);
          delRange.setEnd(node, range.startOffset);
          selection.removeAllRanges();
          selection.addRange(delRange);
          document.execCommand("delete");
          document.execCommand("insertUnorderedList");
          notificarAlteracao();
          return;
        }

        // Checklist ([] ou [ ])
        if (trimmed === "[]" || trimmed === "[ ]") {
          e.preventDefault();
          const startIdx = textBefore.lastIndexOf(trimmed);
          const delRange = document.createRange();
          delRange.setStart(node, startIdx);
          delRange.setEnd(node, range.startOffset);
          selection.removeAllRanges();
          selection.addRange(delRange);
          document.execCommand("delete");
          inserirItemTodo();
          return;
        }
      }
    }

    // 2. Comportamento do Enter dentro de itens de Checklist
    if (e.key === "Enter") {
      const selection = window.getSelection();
      if (!selection || !selection.rangeCount) return;
      const range = selection.getRangeAt(0);
      const targetElement = range.startContainer.nodeType === Node.ELEMENT_NODE
        ? range.startContainer
        : range.startContainer.parentElement;

      const currentTodo = targetElement?.closest(".todo-item");
      if (currentTodo) {
        const textSpan = currentTodo.querySelector(".todo-text");
        const textoAtual = textSpan ? textSpan.textContent.replace(/\u00a0/g, "").trim() : "";

        // Se deu Enter numa linha de tarefa vazia, sai do modo todo
        if (!textoAtual) {
          e.preventDefault();
          currentTodo.remove();
          document.execCommand("insertParagraph");
          notificarAlteracao();
          return;
        }

        // Se tinha texto, cria outro item de checklist logo abaixo
        e.preventDefault();
        inserirItemTodo();
      }
    }
  });

  // Alternar Checkbox com Efeito Riscado e Persistência no HTML
  function alternarCheck(cb) {
    const item = cb.closest(".todo-item");
    if (!item) return;

    if (cb.checked) {
      item.classList.add("completed");
      cb.setAttribute("checked", "checked");
    } else {
      item.classList.remove("completed");
      cb.removeAttribute("checked");
    }
    notificarAlteracao();
  }

  body.addEventListener("change", (e) => {
    if (e.target.classList.contains("todo-check")) {
      alternarCheck(e.target);
    }
  });

  body.addEventListener("click", (e) => {
    if (e.target.classList.contains("todo-check")) {
      alternarCheck(e.target);
    }
  });

  body.addEventListener("input", notificarAlteracao);

  return {
    elemento: container,
    obterConteudo: () => body.innerHTML,
    definirConteudo: (html) => {
      body.innerHTML = html || "";
    },
    focar: () => body.focus()
  };
}
